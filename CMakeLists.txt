cmake_minimum_required(VERSION 3.20)
project(PetrifiedStockfish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PETRIFIED_BMI2 "Build for x86-64 BMI2/AVX2" OFF)
option(PETRIFIED_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(PETRIFIED_PYTHON "Build the pyffish-compatible Python module" OFF)
find_package(Threads REQUIRED)
set(RULES_SOURCES src/attacks.cpp src/bitboard.cpp src/misc.cpp src/movegen.cpp
  src/position.cpp src/memory.cpp src/uci_move.cpp src/petrified_init.cpp
  src/api/petrified_board.cpp src/api/petrified_fen.cpp
  src/api/petrified_notation.cpp src/api/petrified_outcome.cpp)
set(CORE_SOURCES
  src/benchmark.cpp src/evaluate.cpp src/movepick.cpp src/search.cpp
  src/thread.cpp src/timeman.cpp src/tt.cpp src/uci.cpp src/ucioption.cpp
  src/tune.cpp src/syzygy/tbprobe.cpp src/nnue/network.cpp src/engine.cpp
  src/score.cpp src/xboard.cpp)
add_library(petrified_rules STATIC ${RULES_SOURCES})
set_target_properties(petrified_rules PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(petrified_rules PUBLIC src)
target_link_libraries(petrified_rules PUBLIC Threads::Threads)
add_library(petrified_core STATIC ${CORE_SOURCES})
target_include_directories(petrified_core PUBLIC src)
target_link_libraries(petrified_core PUBLIC petrified_rules)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_compile_definitions(petrified_rules PUBLIC IS_64BIT)
endif()
if(NOT MSVC)
  target_compile_options(petrified_rules PRIVATE -Wall -Wextra -Wshadow)
  target_compile_options(petrified_core PRIVATE -Wall -Wextra -Wshadow)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(petrified_rules PRIVATE -fconstexpr-ops-limit=500000000)
    target_compile_options(petrified_core PRIVATE -fconstexpr-ops-limit=500000000)
  endif()
endif()
if(PETRIFIED_BMI2)
  target_compile_definitions(petrified_rules PUBLIC USE_PEXT USE_POPCNT USE_AVX2 USE_SSE2 USE_SSSE3 USE_SSE41)
  target_compile_options(petrified_rules PUBLIC -mavx2 -mbmi2 -mpopcnt)
endif()
if(PETRIFIED_SANITIZERS)
  target_compile_options(petrified_rules PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(petrified_rules PUBLIC -fsanitize=address,undefined)
endif()
if(PETRIFIED_PYTHON)
  find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
  Python3_add_library(pyffish MODULE src/pyffish.cpp WITH_SOABI)
  target_link_libraries(pyffish PRIVATE petrified_rules)
endif()
add_executable(petrified-stockfish src/main.cpp)
target_link_libraries(petrified-stockfish PRIVATE petrified_core)
add_executable(petrified-state-test tests/petrified/state.cpp)
target_link_libraries(petrified-state-test PRIVATE petrified_core)
add_executable(petrified-head-export training/points_head/export.cpp)
target_link_libraries(petrified-head-export PRIVATE petrified_core)
enable_testing()
add_test(NAME state COMMAND petrified-state-test)
